## Example

```python {12}
from signalwire import SWMLService
from signalwire.core.function_result import FunctionResult

service = SWMLService(name="pricing", route="/pricing")
service.add_verb("answer", {})
service.add_verb("hangup", {})

def lookup_competitor(args, raw_data=None):
    competitor = args.get("competitor", "unknown")
    return FunctionResult(f"{competitor} charges $99 per seat.")

service.define_tool(
    name="lookup_competitor",
    description="Look up a competitor's pricing by company name",
    parameters={
        "type": "object",
        "properties": {
            "competitor": {
                "type": "string",
                "description": "Competitor's company name"
            }
        }
    },
    handler=lookup_competitor,
    required=["competitor"]
)
service.serve()
```
