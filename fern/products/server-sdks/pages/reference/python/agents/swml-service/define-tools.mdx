## Example

```python {16}
from signalwire import SWMLService
from signalwire.core.function_result import FunctionResult

service = SWMLService(name="pricing", route="/pricing")

def lookup_competitor(args, raw_data=None):
    return FunctionResult("ACME charges $99 per seat.")

service.define_tool(
    name="lookup_competitor",
    description="Look up a competitor's pricing by company name",
    parameters={"type": "object", "properties": {}},
    handler=lookup_competitor
)

for tool in service.define_tools():
    name = tool["function"] if isinstance(tool, dict) else tool.name
    print(name)
```
