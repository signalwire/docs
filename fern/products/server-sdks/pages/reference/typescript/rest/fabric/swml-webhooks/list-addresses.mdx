List addresses associated with a SWML webhook resource.

<EndpointSchemaSnippet endpoint="GET /api/fabric/resources/swml_webhooks/{swml_webhook_id}/addresses" />

## **Response Example**

<EndpointResponseSnippet endpoint="GET /api/fabric/resources/swml_webhooks/{swml_webhook_id}/addresses" />

## **Example**

```typescript {9-11}
import { RestClient } from "@signalwire/sdk";

const client = new RestClient({
  project: "your-project-id",
  token: "your-api-token",
  host: "your-space.signalwire.com",
});

const addresses = (await client.fabric.swmlWebhooks.listAddresses("webhook-id")) as {
  data?: { id: string; display_name: string }[];
};
for (const addr of addresses.data ?? []) {
  console.log(`${addr.display_name}: ${addr.id}`);
}
```
